512struct LoadStoreId {
514 GlobalThreadId thread_gid;
516 bool initial_write{false};
518 uint64_t pc = 0;
520 uint64_t initial_write_id = 0;
522 uint64_t virtual_id =
523 0;
525 unsigned ppo_rule{0};
526
528 bool is_virtual() const { return virtual_id > 0; }
529
531 bool is_real() const { return virtual_id == 0; }
532
533 // Equality operator: two nodes are equal if they have the same identity
534 // - Real nodes are identified by (thread_gid, pc) for normal instructions, or
535 // by initial_write_id for initial writes
536 // - Virtual nodes are identified by their unique virtual_id (they are created
537 // independently)
538 bool operator==(const LoadStoreId &other) const {
539 // If both are virtual nodes, compare by virtual_id (globally unique)
540 if (is_virtual() && other.is_virtual()) {
541 return virtual_id == other.virtual_id;
542 }
543 // If both are real nodes
544 if (is_real() && other.is_real()) {
545 // For initial writes, compare by initial_write_id (globally unique)
546 if (initial_write && other.initial_write) {
547 return initial_write_id == other.initial_write_id;
548 }
549 // For normal instructions, compare by thread_gid and pc
550 if (!initial_write && !other.initial_write) {
551 return thread_gid == other.thread_gid && pc == other.pc;
552 }
553 // One is initial write, one is not - they are not equal
554 return false;
555 }
556 // Different types (one virtual, one real) are never equal
557 return false;
558 }
560 std::string to_string() const {
561 std::stringstream ss;
562 ss << "(";
563 if (virtual_id) {
564 ss << "Virtual " << virtual_id << " ";
565 }
566 ss << "Warp GID: " << thread_gid.warp_gid
567 << " Thread: " << thread_gid.thread_lid;
568 if (initial_write) {
569 ss << " Initial Write ID: " << initial_write_id;
570 } else {
571 ss << " PC: " << fmt::format("{:#x}", pc);
572 }
573 if (ppo_rule)
574 ss << " Rule: " << ppo_rule;
575 ss << ")";
576 return ss.str();
577 }

580struct LoadStoreIdHash {
581 std::size_t operator()(const LoadStoreId &k) const {
582 // Use type-specific magic constants to avoid collision between different
583 // node types
584 static constexpr std::size_t VIRTUAL_MAGIC = 0x9e3779b97f4a7c15ULL;
585 static constexpr std::size_t INITIAL_WRITE_MAGIC = 0xbf58476d1ce4e5b9ULL;
586 static constexpr std::size_t NORMAL_INSTR_MAGIC = 0x94d049bb133111ebULL;
587
588 // For virtual nodes, mix in VIRTUAL_MAGIC to differentiate from other types
589 if (k.is_virtual()) {
590 return std::hash<uint64_t>()(k.virtual_id ^ VIRTUAL_MAGIC);
591 }
592 // For real nodes
593 if (k.initial_write) {
594 // For initial writes, mix in INITIAL_WRITE_MAGIC to differentiate from
595 // other types
596 return std::hash<uint64_t>()(k.initial_write_id ^ INITIAL_WRITE_MAGIC);
597 }
598 // For normal instructions, mix in NORMAL_INSTR_MAGIC to differentiate from
599 // other types
600 return (GlobalThreadIdHash()(k.thread_gid) ^
601 (std::hash<uint64_t>()(k.pc ^ NORMAL_INSTR_MAGIC) << 1));
602 }

810 static_assert(std::is_trivially_copyable_v<Header>,
811 "Entry must be trivially copyable");
812
813public:
815 void serialize(std::ostream &os) const {
816 Header header;
817 header.magic = Header::MAGIC;
818 header.num_barriers = barriers.size();
819 os.write(reinterpret_cast<const char *>(&header), sizeof(header));
820 for (const auto &[warp_gid, pc_map] : barriers) {
821 os.write(reinterpret_cast<const char *>(&warp_gid), sizeof(warp_gid));
822 uint64_t num_entries = pc_map.size();
823 os.write(reinterpret_cast<const char *>(&num_entries), sizeof(num_entries));
824 for (const auto &[pc, entry] : pc_map) {
825 os.write(reinterpret_cast<const char *>(&pc), sizeof(pc));
826 os.write(reinterpret_cast<const char *>(&entry), sizeof(entry));
827 }
828 }
829 }
830
832 void deserialize(std::istream &is) {
833 Header header;
834 is.read(reinterpret_cast<char *>(&header), sizeof(header));
835 if (header.magic != Header::MAGIC) {
836 throw std::runtime_error("Invalid BarrierMeta data: bad magic");
837 }
838 barriers.clear();
839 for (uint32_t i = 0; i < header.num_barriers; ++i) {
840 WarpGID warp_gid;
841 is.read(reinterpret_cast<char *>(&warp_gid), sizeof(warp_gid));
842 uint64_t num_entries;
843 is.read(reinterpret_cast<char *>(&num_entries), sizeof(num_entries));
844 auto &pc_map = barriers[warp_gid];
845 for (uint64_t j = 0; j < num_entries; ++j) {
846 uint64_t pc;
847 Entry entry;
848 is.read(reinterpret_cast<char *>(&pc), sizeof(pc));
849 is.read(reinterpret_cast<char *>(&entry), sizeof(entry));
850 pc_map[pc] = entry;
851 }
852 }
853 }
854};
