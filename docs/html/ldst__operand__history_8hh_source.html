26struct VGPRValues {
28 std::vector<uint64_t> values;
30 std::vector<bool> tmask;
32 uint32_t warp_id{0};
33
35 bool exact_match(const VGPRValues &other) const {
36 if (tmask.size() != other.tmask.size())
37 return false;
38 for (size_t i = 0; i < values.size() && i < other.values.size(); ++i) {
39 if (i < tmask.size() && tmask[i]) {
40 if (!other.tmask[i] || values[i] != other.values[i])
41 return false;
42 }
43 }
44 return true;
45 }
46
48 bool unordered_partial_match(const VGPRValues &other, bool must_cross_thread = true) const {
49 // Collect active address values from both
50 std::vector<uint64_t> this_addrs, other_addrs;
51 for (size_t i = 0; i < tmask.size(); ++i) {
52 if (tmask[i] && i < values.size())
53 this_addrs.push_back(values[i]);
54 }
55 for (size_t i = 0; i < other.tmask.size(); ++i) {
56 if (other.tmask[i] && i < other.values.size())
57 other_addrs.push_back(other.values[i]);
58 }
59
60 // Same warp: order must be different (reject if identical order)
61 if (must_cross_thread && warp_id == other.warp_id && this_addrs == other_addrs) {
62 return false;
63 }
64
65 // Compare as multiset (sorted)
66 std::sort(this_addrs.begin(), this_addrs.end());
67 std::sort(other_addrs.begin(), other_addrs.end());
68 // Partial equality: truncate to smaller size
69 const auto trunc_sz = std::min(this_addrs.size(), other_addrs.size());
70 this_addrs.resize(trunc_sz);
71 other_addrs.resize(trunc_sz);
72 return this_addrs == other_addrs;
73 }
74
76 bool operator==(const VGPRValues &other) const { return exact_match(other); }
77};

170 template <typename RNG>
171 std::vector<const LoadStoreOperandHistory *>
172 sample_by_type(LoadStoreOperandPairType pair_type, size_t max_samples,
173 RNG &rng,
174 std::optional<uint32_t> exclude_warp_id = std::nullopt) const {
175 auto type_it = type_warp_to_histories_.find(pair_type);
176 if (type_it == type_warp_to_histories_.end()) {
177 return {};
178 }
179 const auto &warp_map = type_it->second;
180
181 // Build list of valid warps and compute total valid count
182 std::vector<std::pair<uint32_t, size_t>> valid_warps; // (warp_id, count)
183 size_t total_valid = 0;
184 for (const auto &[warp_id, entries] : warp_map) {
185 if (!exclude_warp_id.has_value() || warp_id != exclude_warp_id.value()) {
186 valid_warps.emplace_back(warp_id, entries.size());
187 total_valid += entries.size();
188 }
189 }
190
191 if (total_valid == 0) {
192 return {};
193 }
194
195 // If total <= max_samples, return all valid entries
196 if (total_valid <= max_samples) {
197 std::vector<const LoadStoreOperandHistory *> result;
198 result.reserve(total_valid);
199 for (const auto &[warp_id, count] : valid_warps) {
200 for (const auto *ptr : warp_map.at(warp_id)) {
201 result.push_back(ptr);
202 }
203 }
204 return result;
205 }
206
207 // Random sampling: generate k unique random indices in [0, total_valid)
208 // Use Floyd's algorithm for O(k) unique sampling
209 std::unordered_set<size_t> selected_indices;
210 selected_indices.reserve(max_samples);
211 for (size_t i = total_valid - max_samples; i < total_valid; ++i) {
212 std::uniform_int_distribution<size_t> dist(0, i);
213 size_t idx = dist(rng);
214 if (!selected_indices.insert(idx).second) {
215 selected_indices.insert(i);
216 }
217 }
218
219 // Map indices to actual entries using cumulative counts
220 std::vector<const LoadStoreOperandHistory *> result;
221 result.reserve(max_samples);
222
223 // Build cumulative prefix for binary search
224 std::vector<size_t> cumulative;
225 cumulative.reserve(valid_warps.size() + 1);
226 cumulative.push_back(0);
227 for (const auto &[warp_id, count] : valid_warps) {
228 cumulative.push_back(cumulative.back() + count);
229 }
230
231 for (size_t idx : selected_indices) {
232 // Binary search to find which warp this index belongs to
233 auto it = std::upper_bound(cumulative.begin(), cumulative.end(), idx);
234 size_t warp_idx = std::distance(cumulative.begin(), it) - 1;
235 size_t local_idx = idx - cumulative[warp_idx];
236 result.push_back(warp_map.at(valid_warps[warp_idx].first)[local_idx]);
237 }
238
239 return result;
240 }
