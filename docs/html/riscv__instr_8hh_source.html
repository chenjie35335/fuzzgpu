115 void set_tgt_addr(uint64_t pc, uint64_t target) { // for jmp and branches
116 assert(meta_->operand_mask & riscv_meta::HAS_TGT_ADDR);
117 assert(!(pc & 1));
118 assert_log(!(target & 1),
119 "Target address must be aligned to 2-byte boundary");
120
121 if (target >= pc) {
122 uint64_t diff = target - pc;
123 assert(diff <=
124 static_cast<uint64_t>(std::numeric_limits<int64_t>::max()));
125 set_imm_unsafe_(static_cast<int64_t>(diff));
126
127 } else {
128 uint64_t diff = pc - target;
129 assert(diff <=
130 static_cast<uint64_t>(std::numeric_limits<int64_t>::max()) + 1);
131 set_imm_unsafe_(-static_cast<int64_t>(diff));
132 }
133 assigned_operands_ |= riscv_meta::HAS_TGT_ADDR;
134 }
