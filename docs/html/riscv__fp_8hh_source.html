20 // test if it's a scalar/vector floating point instruction
21 const auto opcode = insn_bits & 0x7f;
22 const auto funct3 = (insn_bits >> 12) & 0x7;
23 switch (opcode) {
24 case riscv::OPCODE_FP:
25 return true;
26
27#if defined(ARCH_VENTUS)

29 const auto funct6 = (insn_bits >> 26) & 0x3f;
30 if ((funct3 == 0b110 || funct3 == 0b100) &&
31 (funct6 == 0b000010 || funct6 == 0b000011)) { // vfexp vftta
32 return true;
33 }
34 } break;
35#elif defined(ARCH_VORTEX)
36 case riscv::OPCODE_custom0: { // custom0

38 if (funct7 == 2) { // vx_wmma
39 return true;

41 } break;
42#endif

44 case riscv::OPCODE_V: { // vector
