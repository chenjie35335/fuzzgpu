226 template <typename Tv>
227 bool legal_imm(Tv val) const {
228 if (imm_align_bits) {
229 if ((val & ((1 << imm_align_bits) - 1)))
230 return false;
231 }
232 if (unsigned_imm()) {
233 typename std::make_unsigned<Tv>::type imm_u = val;
234 auto zext = uint64_t(imm_u);
235 return zext < (1ull << imm_len);
236
237 } else {
238 typename std::make_signed<Tv>::type imm_s = val;
239 auto sext = int64_t(imm_s);
240 return sext >= -(1ll << (imm_len - 1)) && sext < (1ll << (imm_len - 1));
241 }
